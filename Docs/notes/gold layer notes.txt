DAHBOARD-Todays Forecast
Loc: Metabase
------------------------------------------------
What are the most important details for a rapid forecast check the day of a hike?
------------------------------------------------
Filters:
	- {{Mountain}} 
		* Field Filter: wiki_mtns.mtn_name
		* Alias Value: wiki_mtns.mtn_name
		* (default value=King's Peak)
		* Dropdown
		* Single select
		* Always require a value
Queries:
	- hi_lo_avg_temps.sql
	- measured_at.sql
select omd.mtn_id,
	   wm.mtn_name,
	   wm.mtn_range,
	   omd.pulled_at,
	   omd.dly_time,
	   omd.measured_at_m,
	   alerts.alert_count,
	   alerts.alert_events,
	   ltng.avg_ltng_potential_j_kg,
	   wc.description,
	   omd.dly_sunrise::time,
	   omd.dly_sunset::time,
	   omd.dly_temperature_2m_min_celsius,
	   omd.dly_temperature_2m_max_celsius,
	   omd.dly_apparent_temperature_min_celsius as feels_like_min_celsius,
	   omd.dly_apparent_temperature_max_celsius as feels_like_max_celsius
-- 	   etc...
from silver.openmeteo_daily as omd
left join silver.wiki_mtns as wm on wm.mtn_id = omd.mtn_id
left join silver.weathercodes as wc on wc.weather_code = omd.dly_weather_code
-- grouping alert events by mtn_id
left join 
(
	select mtn_id,
	count(*) as alert_count,
	string_agg(alert_event, ', ') as alert_events
	from silver.openweather_alerts
	where alert_end >= now()
	group by mtn_id
) as alerts on alerts.mtn_id = omd.mtn_id
-- getting average lightning potential for each mtn_id on current day 
left join 
(
	select mtn_id, 
	ltng_time::date, 
	round(avg(coalesce(ltng_potential_j_kg, 0)),2) as avg_ltng_potential_j_kg
	from silver.openmeteo_lightning
	where ltng_time::date = current_date
	group by mtn_id, ltng_time::date
) as ltng on ltng.mtn_id = omd.mtn_id
where dly_time = current_date
order by mtn_id;
